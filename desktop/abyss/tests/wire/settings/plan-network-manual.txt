plan
network "em0" manual 10.0.0.5/24 router=10.0.0.1 dns=1.1.1.1,9.9.9.9
steps
1. set ifconfig_em0="inet 10.0.0.5 netmask 255.255.255.0" in rc.conf
   $ sysrc -f /etc/rc.conf ifconfig_em0=inet 10.0.0.5 netmask 255.255.255.0
2. set defaultrouter="10.0.0.1" in rc.conf
   $ sysrc -f /etc/rc.conf defaultrouter=10.0.0.1
3. set name_servers="1.1.1.1 9.9.9.9" in resolvconf.conf
   $ sysrc -f /etc/resolvconf.conf name_servers=1.1.1.1 9.9.9.9
4. restart the netif service for em0
   $ service netif restart em0
5. restart the routing service
   $ service routing restart
6. run resolvconf -u
   $ resolvconf -u
